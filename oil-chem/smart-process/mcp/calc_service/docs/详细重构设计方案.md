## 二、步骤 1：创建 CalcContext，缓存从 scenario 剥离

### 2.1 新建 `models/calc_context.py`

```python
# -*- coding: utf-8 -*-
"""计算缓存容器 — 由 Service 层创建和管理生命周期。

设计原则：
  - CalcContext 是可变对象，承载计算过程中的缓存
  - 由 Service 层创建，显式传递给 calculation 层
  - calculation 层可读写，但不负责创建
  - 生命周期与一次求解的 PASS 绑定（或跨 PASS 共享）
"""
from dataclasses import dataclass, field
from typing import Dict, List, Optional, Set


@dataclass
class CalcContext:
    """计算缓存容器"""

    # ── 拓扑缓存（direct_calculator 写入）──
    device_order: Optional[List[str]] = None
    """BFS 拓扑序，首次 calculate_physical 时构建"""

    feed_ratio_cache: Dict[str, Dict[str, float]] = field(default_factory=dict)
    """进料配比缓存 {device_id: {product_id: ratio}}"""

    main_feed_yr_cache: Dict[str, float] = field(default_factory=dict)
    """主进料收率缓存 {device_id: yield_rate}"""

    target_flow_map_cache: Dict[str, List] = field(default_factory=dict)
    """目标流量映射缓存 {device_id: [(product, target_device, ...)]}"""

    # ── 经济缓存（economics / revenue_calculator / cost_calculator 写入）──
    price_table: Dict[str, Dict[str, float]] = field(default_factory=dict)
    """按月分桶价格表 {plan_month: {product_id: price}}"""

    cdu_product_classification: Optional[dict] = None
    """CDU 侧线分类结果"""

    device_cost_cache: Dict[str, Dict[int, float]] = field(default_factory=dict)
    """按月分桶装置成本 {plan_month: {device_backend_id: cost}}"""

    tank_mapping_cache: Optional[Dict[str, str]] = None
    """装置→罐映射 {device_id: tank_id}"""

    tank_flow_mapping_cache: Optional[Dict] = None
    """罐流量映射 {(device_id, product_id): tank_id}"""

    resolve_total_feed_cache: Dict[str, float] = field(default_factory=dict)
    """总进料缓存 {cache_key: total_feed}"""

    # ── 跨组合缓存（由 Service 层管理，calculation 层只读）──
    physical_cache: Optional[Dict] = None
    """跨组合物理计算缓存，key: (batch_id, mode, hangmei_window)"""

    hangmei_precompute_cache: Optional[Dict] = None
    """跨组合航煤预计算缓存，key: (batch_id, mode)"""

    econ_cache: Optional[Dict] = None
    """PASS 4 跨组合经济缓存"""

    def get_price(self, plan_month: str, product_id: str) -> Optional[float]:
        """从 price_table 获取价格"""
        return self.price_table.get(plan_month, {}).get(product_id)

    def set_price(self, plan_month: str, product_id: str, price: float) -> None:
        """写入 price_table"""
        self.price_table.setdefault(plan_month, {})[product_id] = price

    def get_device_cost(self, plan_month: str, device_backend_id: int) -> Optional[float]:
        """从 device_cost_cache 获取装置成本"""
        return self.device_cost_cache.get(plan_month, {}).get(device_backend_id)

    def set_device_cost(self, plan_month: str, device_backend_id: int, cost: float) -> None:
        """写入 device_cost_cache"""
        self.device_cost_cache.setdefault(plan_month, {})[device_backend_id] = cost
```

### 2.2 修改 `models/refinery.py`

**目标**：移除被计算层寄生的 10 个缓存属性，保留模型自身的派生属性缓存。

**保留**（模型派生数据，只读性质，首次访问构建）：
- `_connections_cache` → `connections` property（:169-188）
- `_split_ratio_cache` → `connection_split_ratios` property（:171-206）
- `_upstream_map_cache` / `_downstream_map_cache` → `get_upstream/downstream_connections`（:173-327）
- `_main_feeds_cache` → `get_main_feeds`（:176-418）
- `_cdu_device_id` / `_hangmei_active_ids` / `_yield_switch_ids` / `_tank_device_ids` / `_intermediate_tank_ids` / `_processing_ids`（:339-404）

> **理由**：这些是 scenario 从自身域数据派生的只读视图，生命周期与 scenario 一致，合理保留。

**移除**（计算层寄生缓存，迁移到 CalcContext）：

| 当前位置 | 字段 | 迁移目标 |
|---------|------|---------|
| direct_calculator.py:93 | `scenario._device_order` | `calc_ctx.device_order` |
| direct_calculator.py:120 | `scenario._feed_ratio_cache` | `calc_ctx.feed_ratio_cache` |
| direct_calculator.py:147 | `scenario._main_feed_yr_cache` | `calc_ctx.main_feed_yr_cache` |
| direct_calculator.py:158 | `scenario._target_flow_map_cache` | `calc_ctx.target_flow_map_cache` |
| economics.py:109 | `scenario._cdu_product_classification` | `calc_ctx.cdu_product_classification` |
| economics.py:256 | `scenario._price_table` | `calc_ctx.price_table` |
| revenue_calculator.py:77 | `scenario._tank_mapping_cache` | `calc_ctx.tank_mapping_cache` |
| revenue_calculator.py:96 | `scenario._tank_flow_mapping_cache` | `calc_ctx.tank_flow_mapping_cache` |
| revenue_calculator.py:124 | `scenario._resolve_total_feed_cache` | `calc_ctx.resolve_total_feed_cache` |
| cost_calculator.py:101 | `scenario._device_cost_cache` | `calc_ctx.device_cost_cache` |

**改动方式**：refinery.py 本身不需要删除字段声明（因为这些是 `hasattr` 动态写入的，不在 dataclass 声明中）。只需在计算层停止写入这些属性即可。

```python
def get_main_feeds(products: Dict[str, Product], device_id: str) -> List[Product]:
    """获取某装置的主料产品列表"""
    return [p for p in products.values()
            if p.material_type == 'main_feed' and p.source_device_id == device_id]

def get_upstream_connections(connections: Dict[str, Connection], device_id: str) -> List[Connection]:
    """获取装置的上游连接"""
    return [c for c in connections.values() if c.to_device_id == device_id]

def get_downstream_connections(connections: Dict[str, Connection], device_id: str) -> List[Connection]:
    """获取装置的下游连接"""
    return [c for c in connections.values() if c.from_device_id == device_id]
```

### 3.3 `calculation/direct_calculator.py` — calculate_direct 签名变更

**当前**（:363-378）：接收 `scenario`，纯透传给 calculate_physical 和 generate_explanation。

```python
def calculate_direct(
    # 域数据
    devices: Dict[str, Device],
    connections: Dict[str, Connection],
    products: Dict[str, Product],
    material_flows: Dict[str, MaterialFlow],
    start_device_id: str,
    cdu_device_id: str,
    crude_type: str,
    processing_device_ids: Set[str],
    yield_switch_device_ids: Set[str],
    hangmei_active_device_ids: Set[str],
    tank_device_ids: Set[str],
    price_costs: Dict[str, Dict[str, float]],
    product_material_map: Dict[str, int],
    material_name_map: Dict[str, int],
    # 计算参数
    input_amount: float,
    yield_mode: str,
    logger=None,
    crude_oil_price: float = 0.0,
    total_input_amount: float = None,
    days: int = 1,
    hangmei_mode: bool = False,
    hangmei_m_days: float = 0,
    day_index: float = 0,
    plan_month: str = None,
    shutdown_intervals: dict = None,
    capacity_only: bool = False,
    summary_only: bool = False,
    # 缓存
    calc_ctx: Optional[CalcContext] = None,
) -> Tuple[bool, dict]:
    physical = calculate_physical(
        devices, connections, products, material_flows,
        start_device_id, cdu_device_id, crude_type,
        processing_device_ids, yield_switch_device_ids,
        hangmei_active_device_ids, tank_device_ids,
        input_amount, yield_mode, logger=logger,
        total_input_amount=total_input_amount, days=days,
        hangmei_mode=hangmei_mode, hangmei_m_days=hangmei_m_days,
        day_index=day_index, shutdown_intervals=shutdown_intervals,
        calc_ctx=calc_ctx,
    )
    if capacity_only:
        return physical.feasible, physical.to_dict()

    explanation = generate_explanation(
        products, devices, connections, material_flows,
        cdu_device_id, processing_device_ids,
        hangmei_active_device_ids,

        day_index=day_index, hangmei_m_days=hangmei_m_days,
        logger=logger,
```

```python
# service/solve_service.py

def _finalize_combination_outputs(
    pass1_results: Dict[str, PassResult],
    combination_results: List[CombinationOutput],
) -> None:
    """步骤 A：从 PassResult 填充 CombinationOutput 的 batch_details 和 monthly_load"""
    for output in combination_results:
        pr = pass1_results[output.combination_id]
        output.batch_details = pr.batch_details
        output.monthly_load = pr.monthly_load


def _reassess_feasibility(
    combination_results: List[CombinationOutput],
) -> None:
    """步骤 B：按月度口径重定可行性（含 near_feasible 标记）"""
    for output in combination_results:
        ml = output.monthly_load
        if not ml:
            continue
        overloaded = [d for d in ml.get('devices', []) if d.get('is_overloaded')]
        if not overloaded:
            output.feasible = True
            output.near_feasible = False
            output.infeasible_summary = ""
        elif len(overloaded) <= 1 and overloaded[0].get('monthly_util', 1) < 1.1:
            output.feasible = True
            output.near_feasible = True
            output.infeasible_summary = f"{overloaded[0]['device_id']}月负荷轻微超容"
        else:
            output.feasible = False
            output.infeasible_summary = "、".join(
                f"{d['device_id']}超容{d.get('overload_amount', 0):.0f}吨"
                for d in overloaded
            )


def _select_optimal(
    combination_results: List[CombinationOutput],
) -> Optional[CombinationOutput]:
    """步骤 C：选最优组合（可行 > near_feasible > 不可行，同级别按 revenue 降序）"""
    feasible = [c for c in combination_results if c.feasible and not c.near_feasible]
    near = [c for c in combination_results if c.feasible and c.near_feasible]
    if feasible:
        return max(feasible, key=lambda c: c.total_revenue)
    if near:
        return max(near, key=lambda c: c.total_revenue)
    return max(combination_results, key=lambda c: c.total_revenue) if combination_results else None
```

### 6.3 device_utilization 深层变异提取

当前 `_enrich` 中修改 `device_utilization[did]['daily_consumption']` 和 `device_utilization[did]['processing_days']`（solve_service.py:1133-1134）的逻辑，提取为独立函数：

```python
def _annotate_device_processing_days(
    batch_details: List[dict],
    monthly_load: dict,
) -> None:
    """在 batch_details 的 device_utilization 中补充 processing_days 和 daily_consumption"""
    for bd in batch_details:
        dev_util = bd.get('device_utilization', {})
        for did, util in dev_util.items():
            ml_device = next((d for d in monthly_load.get('devices', []) if d['device_id'] == did), None)
            if ml_device and ml_device.get('is_overloaded'):
                util['daily_consumption'] = ml_device['daily_capacity']
                util['processing_days'] = ml_device['monthly_input'] / ml_device['daily_capacity']
```
